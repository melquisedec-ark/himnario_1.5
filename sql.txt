CREATE DATABASE himnario_web CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;

USE himnario_web;

-- Tabla para la información general del himno
CREATE TABLE himnos (
    id INT AUTO_INCREMENT PRIMARY KEY,
    numero INT NOT NULL UNIQUE, -- El número clásico del himnario
    titulo VARCHAR(255) NOT NULL,
    creado_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Tabla para el contenido (letras)
CREATE TABLE estrofas (
    id INT AUTO_INCREMENT PRIMARY KEY,
    himno_id INT NOT NULL,
    tipo ENUM('verso', 'coro', 'puente') NOT NULL DEFAULT 'verso',
    orden INT NOT NULL, -- Para saber qué va primero (1, 2, 3...)
    contenido TEXT NOT NULL,
    FOREIGN KEY (himno_id) REFERENCES himnos(id) ON DELETE CASCADE
);
